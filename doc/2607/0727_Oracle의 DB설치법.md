# 로그인 방법

초반 설치를할때, 이런게 잠깐 지나갔는데... 주의깊게 봤다. 

SYS, SYSTEM, PDBADMIN 세가지 계정을 기본 생성한다. 

이때, SYS계정은 root 계정과도 같이 절대적인 권한을 가진 계정이다. 

sys_bluelagoon , 롤 SYSDBA
SYS
shorgsage1019	
서비스이름 : FREEPDB1

를 염두해두고 


CREATE USER SHORTS_DB IDENTIFIED BY sys_bluelagoon; #sys_bluelagoon이라는 식별자값을 사용한 DB를 만든다. 

GRANT CREATE SESSION TO SHORTS_DB; # Oracle DB에 로그인할 수 있는 권한.
GRANT CREATE TABLE TO SHORTS_DB; # GRANT CREATE TABLE TO SHORTS_DB;
GRANT CREATE SEQUENCE TO SHORTS_DB;

ALTER USER SHORTS_DB QUOTA UNLIMITED ON USERS;

이런식의 쿼리를 만든다. 


CREATE TABLE MEMBER (
    MEMBER_ID NUMBER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
    USERNAME VARCHAR2(50) NOT NULL,
    CREATED_AT TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);
