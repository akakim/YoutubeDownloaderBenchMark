CREATE TABLE MEMBER (

    
    MEMBER_ID NUMBER GENERATED BY DEFAULT AS IDENTITY
        PRIMARY KEY,

    USERNAME VARCHAR2(50)
        NOT NULL,

    CI_VALUE VARCHAR2(200)
        NOT NULL,

    AUTHORITY VARCHAR2(30)
        DEFAULT 'ROLE_USER'
        NOT NULL,

    EMAIL VARCHAR2(255)
        NOT NULL,

    PASSWORD_HASH VARCHAR2(255)
        NOT NULL,

    CREATED_AT TIMESTAMP
        DEFAULT CURRENT_TIMESTAMP
        NOT NULL,

    CONSTRAINT UK_MEMBER_USERNAME UNIQUE (USERNAME),
    CONSTRAINT UK_MEMBER_CI UNIQUE (CI_VALUE),
    CONSTRAINT UK_MEMBER_EMAIL UNIQUE (EMAIL),

    CONSTRAINT CK_MEMBER_AUTHORITY CHECK (
        AUTHORITY IN ('ROLE_USER', 'ROLE_ADMIN')
    )
);
